function(echo_add_update_helper target)
    add_executable(${target} main.cpp)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(WIN32)
        target_link_libraries(${target} PRIVATE bcrypt shell32)
        set_property(
            TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
        )
    elseif(APPLE)
        target_compile_options(${target} PRIVATE -Wno-deprecated-declarations)
    else()
        find_package(OpenSSL REQUIRED COMPONENTS Crypto)
        target_link_libraries(${target} PRIVATE OpenSSL::Crypto)
    endif()
endfunction()
echo_add_update_helper(echo-update-helper)
if(BUILD_TESTING)
    echo_add_update_helper(echo-update-helper-test-probe)
    target_compile_definitions(echo-update-helper-test-probe PRIVATE ECHO_UPDATE_HELPER_TESTING=1)
endif()
